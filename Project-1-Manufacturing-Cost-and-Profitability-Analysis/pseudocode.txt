START

INPUT units
INPUT material_cost_per_unit
INPUT labour_cost_per_unit
INPUT wastage_cost
INPUT machine_cost
INPUT selling_price_per_unit
INPUT target_profit

IF units <= 0 OR any cost, price, or target is negative THEN
    DISPLAY "Invalid input. Enter valid values."
ELSE
    material_cost = units * material_cost_per_unit
    labour_cost = units * labour_cost_per_unit
    total_cost = material_cost + labour_cost + wastage_cost + machine_cost
    revenue = units * selling_price_per_unit
    profit = revenue - total_cost
    cost_per_unit = total_cost / units

    DISPLAY material_cost
    DISPLAY labour_cost
    DISPLAY wastage_cost
    DISPLAY machine_cost
    DISPLAY total_cost
    DISPLAY cost_per_unit
    DISPLAY revenue
    DISPLAY profit

    IF profit >= target_profit THEN
        DISPLAY "Profit Target Achieved"
    ELSE
        DISPLAY "Profit Target Not Achieved"
    END IF
END IF

STOP
